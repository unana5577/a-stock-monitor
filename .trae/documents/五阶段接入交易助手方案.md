### 3. ui_m1.js 修改（约 30 行）

在 `fetchOverview()` 或新增独立 `fetchStageState()` 中：

```js
// 新增 ref
const stageMap = ref({})       // { symbol: { stage, stage_icon, ma20, ma60, ... } }
const marketState = ref('--')
const marketConfidence = ref(0)

// 新增 fetch
async function fetchStageState() {
  const r = await fetch('/api/m1/stage_state')
  const j = await r.json()
  if (j.ok) {
    stageMap.value = j.data.stages
  }
}
```

在 `tradeTableRows` computed 中追加：

```js
const st = stageMap.value[r.symbol]
if (st) {
  row.stage = st.stage
  row.stageIcon = st.stage_icon
}
```
